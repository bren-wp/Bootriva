cmake_minimum_required(VERSION 3.20)
project(Bootriva VERSION 0.2.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Keep the CRT consistent across the executable, core static library and tests.
# /MT in Release avoids dependency on an external Visual C++ Redistributable.
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(bootriva_core STATIC src/core.cpp)
target_include_directories(bootriva_core PUBLIC ${PROJECT_SOURCE_DIR}/include)
if(MSVC)
  target_compile_options(bootriva_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(bootriva_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
enable_testing()
add_executable(bootriva_tests tests/core_tests.cpp)
target_link_libraries(bootriva_tests PRIVATE bootriva_core)
add_test(NAME core_tests COMMAND bootriva_tests)
if(WIN32)
  enable_language(RC)
  configure_file(src/win/bootriva.rc.in "${CMAKE_CURRENT_BINARY_DIR}/bootriva.rc" @ONLY)
  add_executable(Bootriva WIN32 src/win/main.cpp src/win/usb_win.cpp "${CMAKE_CURRENT_BINARY_DIR}/bootriva.rc")
  target_link_libraries(Bootriva PRIVATE bootriva_core bcrypt dwmapi comctl32 comdlg32 shell32)
  target_compile_definitions(Bootriva PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  if(MSVC)
    target_compile_options(Bootriva PRIVATE /W4 /permissive- /utf-8)
    set_property(TARGET Bootriva PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_link_options(Bootriva PRIVATE /MANIFEST:NO)
  endif()
  set_target_properties(Bootriva PROPERTIES OUTPUT_NAME "Bootriva")
  # This target deliberately has no embedded requireAdministrator manifest.
  # It exercises the real window/control creation path on CI and catches the
  # "empty window" regression without touching a USB disk.
  add_executable(bootriva_ui_smoke src/win/main.cpp src/win/usb_win.cpp)
  target_compile_definitions(bootriva_ui_smoke PRIVATE
    BOOTRIVA_UI_SMOKE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(bootriva_ui_smoke PRIVATE
    bootriva_core bcrypt dwmapi comctl32 comdlg32 shell32)
  if(MSVC)
    target_compile_options(bootriva_ui_smoke PRIVATE /W4 /permissive- /utf-8)
    set_property(TARGET bootriva_ui_smoke PROPERTY
      MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
  add_test(NAME win32_gui_controls COMMAND bootriva_ui_smoke)
  set_tests_properties(win32_gui_controls PROPERTIES TIMEOUT 45)
  install(TARGETS Bootriva RUNTIME DESTINATION .)
endif()
